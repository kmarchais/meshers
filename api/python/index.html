def generate(  # ruff: ignore[complex-structure, too-many-branches]
    field: str | Field | CompiledField,
    *,
    gradient: Field | None = None,
    bounds: Sequence[float] = (0.0, 1.0, 0.0, 1.0, 0.0, 1.0),
    cells: int | Sequence[int] = 24,
    periodic: Sequence[bool] = (False, False, False),
    band: tuple[float, float] | None = None,
    geometry_tolerance: float = 0.01,
    minimum_quality: float = 0.0,
    max_tetrahedra: int = 2_000_000,
    optimize_passes: int = 4,
    interior_refine_nodes: int = 0,
    snap: float = 0.2,
    threads: int | None = 1,
    batch_size: int = 4096,
    cancel: CancellationToken | None = None,
    compile: bool = True,
    adapt: bool = False,
    min_length: float | None = None,
    max_length: float | None = None,
    gradation: float = 0.3,
    layers: float = 0.0,
    adapt_iterations: int = 4,
    quality_guard: float = 0.3,
    adapt_tolerance: float | None = None,
    order: int = 1,
    size_field: Field | None = None,
    curved_fit: bool = False,
    expected_components: int | None = None,
) -> Mesh:
    """Generate a tetrahedral mesh directly from a bounded implicit field.

    Args:
        expected_components: Require this many face-connected solid components in the cut box,
            before pairing periodic faces. Zero requires sampled-empty output.
        field: Pointwise function of physical x, y, z coordinates, or a built-in
            name: ``gyroid``, ``schwarz_p``, ``schwarz_d``. Built-ins have period 1.
        gradient: Optional gradient of the complete field, including grading.
            Traced gradients return three expressions; NumPy callbacks return (N,3).
        bounds: ``(xmin, xmax, ymin, ymax, zmin, zmax)`` in physical units.
        cells: Background lattice intervals, 4..128 per axis. An int applies to all axes.
        periodic: Explicit matching constraints for x, y, z. The field must match
            across requested faces; periodicity is not inferred.
        band: Keep ``lower < field < upper``. Without a band, keep the negative side.
        geometry_tolerance: Maximum sampled boundary-distance estimate, in physical units.
        minimum_quality: Required minimum MMG tetrahedron quality, 0..1. Zero disables
            the requirement. Adaptive splitting and wall projection honor a positive
            requested minimum. Nonempty meshes below it are rejected, and requested
            geometry limits are not relaxed. Empty meshes remain valid.
        max_tetrahedra: Budget for generated tetrahedra. Adaptation raises
            MeshingError when a local topology operation exceeds this budget.
        optimize_passes: Number of quality-optimization passes, 0..100.
        interior_refine_nodes: Maximum new nodes in one length-ranked interior
            bisection pass after scalar surface generation or adaptation. Zero
            disables it. For order=2, this counts new corner nodes; quadratic
            edge nodes are additional. Boundary vertices and triangles stay
            fixed. Disjoint cavities, the tetrahedron budget and a quality floor
            of max(minimum_quality, 0.1) may limit the number inserted.
        snap: Maximum lattice snap as a fraction of minimum spacing, 0..0.2.
        threads: Native worker limit, default one. None uses available CPU parallelism, capped
            at 256. One uses one worker; zero retains legacy serial update ordering.
        batch_size: Maximum NumPy callback batch size.
        cancel: Optional cooperative cancellation token.
        compile: Compile supported expressions and their first derivatives
            automatically. False selects NumPy callbacks. Unsupported operations
            emit a warning and use callbacks, which must return float64 (N,) arrays.
            Eligible Tet4 convex polyhedra retain exact half-space constraints;
            requested periodic axes must leave every plane invariant. Diagnostics
            report ``polyhedral_constraints``. Bounded min/max combinations of
            convex affine fields retain nonconvex regions on the nonperiodic path.
            Eligible nonperiodic outer-max intersections retain analytic branches;
            diagnostics report ``retained_constraints``.
        adapt: Experimental. Replace fixed surface refinement by size-driven
            adaptation: boundary edges follow the curvature needed for
            ``geometry_tolerance``, the interior grades towards ``max_length``.
            ``cells`` then only sets the starting lattice, which must resolve topology.
            Compiled convex polyhedra preserve their exact planes and retain
            existing crease vertices. Face/interior coarsening, splitting,
            spatial sizing and constrained quality improvement remain active.
        min_length: Smallest adapted edge length; default ``geometry_tolerance``.
        max_length: Largest adapted edge length; default a quarter of the smallest box side.
        gradation: Maximum length growth per unit distance (MMG ``hgrad - 1``).
        layers: For bands, minimum element count across the local sheet thickness.
        adapt_iterations: Split/collapse/swap/relocate rounds.
        quality_guard: Operations may not create MMG quality below this value
            unless the affected elements were already worse.
        adapt_tolerance: Distance the adapter aims for; default ``geometry_tolerance``,
            which remains the acceptance check.
        order: 1 for linear tetrahedra, 2 for 10-node tetrahedra (VTK/Gmsh/fedoo
            node order) with wall mid-edge nodes projected onto the field where
            sampled validity permits. Invalid corrections are reduced or straightened;
            ``diagnostics["quadratic"]`` records both counts. Quadratic meshes
            report ``tetrahedra`` of shape (M, 10) and ``surface`` of shape (K, 6);
            ``periodic_pairs`` then include cap midpoint nodes and preserve
            complete constraint classes for existing periodic node images.
        size_field: Optional vectorized callback returning positive physical target
            edge lengths, shape (N,), for ``adapt=True``. Targets combine with
            curvature sizing by taking the smaller length, then applying
            ``min_length``, ``max_length`` and gradation. Opposite periodic traces
            must match. This callback is evaluated in batches without tracing.
        curved_fit: Experimental bounded fitting of the final quadratic wall faces.
            Requires order=2. Corners and connectivity stay fixed; wall edge nodes
            may move off the exact implicit surface to improve the complete patch.
            Cap coordinates and periodic aliases stay coupled. Sampled validity
            guards each accepted move. This changes the discrete geometry and FE
            space; it does not transfer an old solution or certify convergence.

    Returns:
        Mesh: Owned arrays and diagnostics. Empty geometries return empty arrays.
            Timing excludes field preparation, which is reported separately.

    Raises:
        ValueError: Options or callback outputs are invalid.
        MeshingError: Meshing fails its geometry or topology checks.
        CancelledError: The cancellation token is triggered.

    Note:
        Fields must be deterministic and independent of batch contents. Tracing
        executes a function once per preparation, so avoid side effects. Finite
        differences need the field slightly beyond the box. Sampling cannot
        certify arbitrarily small features. Native threads release the Python GIL;
        Python fallback callbacks execute on the calling thread.
    """
    if expected_components is not None and (
        isinstance(expected_components, (bool, np.bool_))
        or not isinstance(expected_components, (int, np.integer))
        or expected_components < 0
    ):
        raise ValueError("expected_components must be a nonnegative integer")

    b = np.asarray(bounds, dtype=np.float64)
    if curved_fit and order != 2:
        raise ValueError("curved_fit requires order 2")
    if size_field is not None and not adapt:
        raise ValueError("size_field requires adapt=True")
    if b.shape != (6,):
        raise ValueError("bounds must contain xmin,xmax,ymin,ymax,zmin,zmax")
    if isinstance(cells, (int, np.integer)):
        cells = (int(cells),) * 3
    if len(periodic) != 3 or any(not isinstance(v, (bool, np.bool_)) for v in periodic):
        raise ValueError("periodic must contain three booleans")
    if threads is None:
        threads = _meshers.available_threads()
    preparation_start = time.perf_counter()
    gradient_override = gradient is not None
    evaluator = (
        "native"
        if isinstance(field, str)
        else "compiled"
        if isinstance(field, CompiledField)
        else "python"
    )
    if callable(field) and not isinstance(field, CompiledField) and compile:
        try:
            field = compile_field(field, gradient=gradient)
            gradient = None
            evaluator = "compiled"
        except TraceError as error:
            warnings.warn(f"Using NumPy callbacks: {error}", RuntimeWarning, stacklevel=2)
    scalar_mode = (
        order != 1,
        band is not None,
        any(periodic) and not _periodic_affine(field, periodic, b if adapt else None),
        threads != 1,
        interior_refine_nodes != 0,
        gradient_override,
        # Nonadaptive SDF retention would bypass the original parent's range checks.
        not adapt and getattr(field, "_sdf_intersection", False),
        not 0 <= optimize_passes <= 20,
        max_tetrahedra <= 0,
        not all(
            isinstance(value, (int, np.integer))
            for value in (threads, interior_refine_nodes, optimize_passes, max_tetrahedra)
        ),
    )
    planes = _affine_planes(field, scalar_mode)
    if (
        planes is None
        and (not adapt or getattr(_meshers, "ADAPTIVE_RETAINED_INTERSECTIONS", False))
        and not any(scalar_mode)
    ):
        planes = _retained_constraints(field, b)
    preparation_seconds = time.perf_counter() - preparation_start
    if planes is not None and not adapt:
        mesh = generate_intersection(
            planes,
            periodic=periodic,
            bounds=b.tolist(),
            cells=cells,
            geometry_tolerance=geometry_tolerance,
            minimum_quality=minimum_quality,
            max_tetrahedra=int(max_tetrahedra),
            optimize_passes=int(optimize_passes),
            snap=snap,
            batch_size=batch_size,
            cancel=cancel,
        )
        mesh.diagnostics.update(
            threads=1,
            evaluator="compiled",
            inserted_interior_nodes=0,
            preparation_seconds=preparation_seconds
            + mesh.diagnostics.get("preparation_seconds", 0.0),
        )
        mesh.diagnostics[
            "polyhedral_constraints"
            if getattr(field, "_halfspaces", None) is not None
            else "retained_constraints"
        ] = len(planes)
        return _check_components(mesh, expected_components)
    feature_options = (
        {"halfspaces": getattr(field, "_halfspaces", None)}
        if planes is not None and adapt and getattr(_meshers, "ADAPTIVE_POLYHEDRAL", False)
        else {}
    )
    feature_options.update(
        {"constraints": list(planes.values())}
        if planes is not None and adapt and getattr(field, "_halfspaces", None) is None
        else {}
    )
    planes = (
        getattr(field, "_polyhedral_regions", None)
        if isinstance(field, CompiledField)
        and not any(scalar_mode)
        and getattr(_meshers, "POLYHEDRAL_REGIONS", False)
        else None
    )
    if planes is not None:
        feature_options = {"halfspaces": planes[0], "regions": planes[1]}
    result = _meshers.generate(
        field,
        gradient,
        [b[::2].tolist(), b[1::2].tolist()],
        cells,
        tuple(bool(v) for v in periodic),
        band,
        geometry_tolerance,
        minimum_quality,
        max_tetrahedra,
        optimize_passes,
        interior_refine_nodes,
        snap,
        threads,
        batch_size,
        cancel,
        (
            geometry_tolerance if min_length is None else float(min_length),
            float(np.min(b[1::2] - b[::2])) / 4 if max_length is None else float(max_length),
            float(gradation),
            float(layers),
            int(adapt_iterations),
            float(quality_guard),
            geometry_tolerance if adapt_tolerance is None else float(adapt_tolerance),
        )
        if adapt
        else None,
        int(order),
        size_field,
        curved_fit=curved_fit,
        **feature_options,
    )
    quadratic = result.pop("quadratic", None)
    if quadratic is not None:
        for name in ("points", "tetrahedra", "surface", "periodic_pairs"):
            result[name] = quadratic.pop(name)
        result["diagnostics"]["quadratic"] = quadratic
    result["diagnostics"]["threads"] = threads
    result["diagnostics"]["evaluator"] = evaluator
    result["diagnostics"]["preparation_seconds"] = preparation_seconds
    result["periodic_pairs"] = tuple(result["periodic_pairs"])
    return _check_components(Mesh(**result), expected_components)

Mesh the intersection of named negative sublevel sets, preserving their curves.

Constraints are evaluated in parameter coordinates. A coordinate map returns a tuple (X(x,y,z), Y(x,y,z), Z(x,y,z)) and acts on the background before cutting; it must preserve orientation and be locally invertible. Straight-sided tetrahedra approximate the mapped geometry. This path currently uses one worker.

def generate_intersection(
    constraints: Mapping[str, Callable | _meshers.CompiledField | str],
    *,
    bounds: Sequence[float] = (0.0, 1.0, 0.0, 1.0, 0.0, 1.0),
    cells: int | Sequence[int] = 24,
    coordinate_map: Callable | None = None,
    periodic: Sequence[bool] = (False, False, False),
    periodic_transforms: Mapping[int, Sequence[Sequence[float]]] | None = None,
    closed_axes: Sequence[bool] = (False, False, False),
    adapt: bool = False,
    inverse_map: Callable | None = None,
    physical_bounds: Sequence[float] | None = None,
    seed_tolerance: float = 0.01,
    min_length: float | None = None,
    max_length: float | None = None,
    gradation: float = 0.3,
    adapt_iterations: int = 4,
    quality_guard: float = 0.3,
    adapt_tolerance: float | None = None,
    geometry_tolerance: float | None = None,
    minimum_quality: float = 0.0,
    optimize_passes: int = 4,
    snap: float = 0.2,
    max_tetrahedra: int = 2_000_000,
    compile: bool = True,
    batch_size: int = 4096,
    cancel: _meshers.CancellationToken | None = None,
) -> Mesh:
    """Mesh the intersection of named negative sublevel sets, preserving their curves.

    Constraints are evaluated in parameter coordinates. A coordinate map returns
    a tuple ``(X(x,y,z), Y(x,y,z), Z(x,y,z))`` and acts on the background before
    cutting; it must preserve orientation and be locally invertible. Straight-sided
    tetrahedra approximate the mapped geometry. This path currently uses one worker.

    Args:
        constraints: Ordered mapping of 2-32 unique nonempty names to fields
            (8 without wide-constraints). Keep
            every field <= 0. Use two constraints for the sides of a TPMS sheet.
        bounds: Parameter box ``(xmin,xmax,ymin,ymax,zmin,zmax)``.
        cells: Background intervals, 4-128 per parameter axis.
        coordinate_map: Optional parameter-to-physical map; default identity.
        periodic: Matching low/high faces along the three parameter axes.
        periodic_transforms: Rigid homogeneous 4x4 matrices keyed by periodic axis
            (0, 1, 2), mapping low to high physical points. Required for each mapped
            periodic axis. Identity maps default to box translations. Node and
            triangle correspondence are checked after optimization.
        closed_axes: Identify coincident periodic caps before final repair. Requires
            identity periodic transforms; closed seams have no external node pairs.
        adapt: Adapt a validated mapped seed in physical space. Requires both maps
            and physical bounds; external periodic faces are not supported.
        inverse_map: Physical-to-parameter coordinates for mapped adaptation.
        physical_bounds: Physical bounding box used for adaptation.
        seed_tolerance: Separate sampled distance limit for the starting mesh.
        min_length: Smallest target length during adaptation.
        max_length: Largest target length during adaptation.
        gradation: Maximum target-length growth per unit distance.
        adapt_iterations: Number of size-driven adaptation rounds.
        quality_guard: Local quality floor for adaptive operations.
        adapt_tolerance: Adaptive distance target; defaults to the final tolerance.
        geometry_tolerance: Optional maximum sampled physical surface-distance
            estimate. None skips this relatively expensive check. This is an
            acceptance gate, not adaptive refinement or a Hausdorff bound.
        minimum_quality: Required final minimum MMG quality, 0-1; zero disables gate.
        optimize_passes: Integrated constrained optimization rounds, 0-20.
        snap: Joint background snap fraction, 0-0.2.
        max_tetrahedra: Output element budget.
        compile: Automatically compile supported field and map expressions.
        batch_size: Maximum callback batch size, 1-65536.
        cancel: Cooperative cancellation token.

    Returns:
        Mesh with physical points, parameter coordinates, constraint masks/names,
        feature edges, validated periodic pairs/transforms and quality diagnostics.

    Raises:
        ValueError: An option, name, transform or callback output is invalid.
        MeshingError: Construction, geometry, topology, quality or periodic checks fail.
        CancelledError: Generation is cancelled.

    Note:
        Constraints remain separate throughout construction. Optimization can move
        vertices on their assigned analytic surfaces and change the facet approximation.
        Acute input angles can make a requested minimum quality unattainable.
    """
    from . import Mesh

    if adapt:
        return _adapt_mapped_intersection(
            constraints,
            bounds=bounds,
            cells=cells,
            mapping=coordinate_map,
            inverse_map=inverse_map,
            physical_bounds=physical_bounds,
            periodic=periodic,
            transforms=periodic_transforms,
            closed_axes=closed_axes,
            seed_tolerance=seed_tolerance,
            tolerance=geometry_tolerance,
            minimum_quality=minimum_quality,
            passes=optimize_passes,
            snap=snap,
            maximum=max_tetrahedra,
            compile=compile,
            batch_size=batch_size,
            cancel=cancel,
            min_length=min_length,
            max_length=max_length,
            gradation=gradation,
            adapt_iterations=adapt_iterations,
            quality_guard=quality_guard,
            adapt_tolerance=adapt_tolerance,
        )
    b, cells, transforms = _validate_options(
        constraints,
        bounds=bounds,
        cells=cells,
        periodic=periodic,
        coordinate_map=coordinate_map,
        periodic_transforms=periodic_transforms,
        optimize_passes=optimize_passes,
        snap=snap,
        minimum_quality=minimum_quality,
        geometry_tolerance=geometry_tolerance,
    )
    if len(closed_axes) != 3 or any(not isinstance(v, (bool, np.bool_)) for v in closed_axes):
        raise ValueError("closed_axes must contain three booleans")
    if any(
        closed_axes[a]
        and (not periodic[a] or not np.allclose(transforms[a], np.eye(4), rtol=0, atol=1e-12))
        for a in range(3)
    ):
        raise ValueError("closed axes require periodic identity transforms")
    start = time.perf_counter()
    fields = [_prepare(f, compile=compile) for f in constraints.values()]
    mapping = None
    if coordinate_map is not None:
        mapping = [
            _prepare(lambda x, y, z, a=a: coordinate_map(x, y, z)[a], compile=compile)
            for a in range(3)
        ]
    preparation = time.perf_counter() - start
    result = _meshers.generate_intersection(
        fields,
        mapping,
        [b[::2].tolist(), b[1::2].tolist()],
        cells,
        tuple(periodic),
        tuple(closed_axes),
        minimum_quality,
        max_tetrahedra,
        optimize_passes,
        snap,
        batch_size,
        cancel,
    )
    _check_cancel(cancel)
    tags, pairs = _periodic_metadata(
        result["points"],
        result["parameters"],
        result["surface"],
        b,
        periodic=tuple(p and not c for p, c in zip(periodic, closed_axes, strict=True)),
        transforms=transforms,
    )
    diagnostics = result["diagnostics"]
    diagnostics["preparation_seconds"] = preparation
    diagnostics["evaluator"] = (
        "compiled" if all(isinstance(f, _meshers.CompiledField) for f in fields) else "mixed"
    )
    diagnostics["empty"] = len(result["tetrahedra"]) == 0
    _refresh_linear_metrics(result, fitted=False)
    diagnostics["sampled_surface_error"] = None
    if geometry_tolerance is not None:
        fields = [_periodic_extension(f, b, closed_axes) for f in fields]
        if mapping is not None:
            mapping = [_periodic_extension(f, b, closed_axes) for f in mapping]
        try:
            errors = _surface_errors(
                result,
                fields,
                mapping,
                b,
                cancel,
                batch_size=batch_size,
                geometry_tolerance=geometry_tolerance,
                closed_axes=closed_axes,
            )
        except np.linalg.LinAlgError as error:
            raise _meshers.MeshingError("singular coordinate map during surface checks") from error
        diagnostics["surface_errors"] = dict(zip(constraints, errors, strict=True))
        diagnostics["sampled_surface_error"] = max(errors)
        if max(errors) > geometry_tolerance and not any(
            p and not c for p, c in zip(periodic, closed_axes, strict=True)
        ):
            fitting_start = time.perf_counter()
            moved = _fit_surface(
                result,
                fields,
                mapping,
                b,
                cells=cells,
                closed_axes=closed_axes,
                tolerance=geometry_tolerance,
                minimum_quality=minimum_quality,
                cancel=cancel,
                batch_size=batch_size,
            )
            diagnostics["geometry_vertex_moves"] = moved
            diagnostics["surface_fitting_seconds"] = time.perf_counter() - fitting_start
            errors = _surface_errors(
                result,
                fields,
                mapping,
                b,
                cancel,
                batch_size=batch_size,
                geometry_tolerance=geometry_tolerance,
                closed_axes=closed_axes,
            )
            diagnostics["surface_errors"] = dict(zip(constraints, errors, strict=True))
            diagnostics["sampled_surface_error"] = max(errors)
            _refresh_linear_metrics(result, fitted=bool(moved))
        if max(errors) > geometry_tolerance:
            raise _meshers.MeshingError(
                f"sampled surface error {max(errors):g} exceeds {geometry_tolerance:g}"
            )
        caps = _mapped_cap_errors(
            result,
            tags,
            mapping,
            b,
            closed_axes=closed_axes,
            geometry_tolerance=geometry_tolerance,
            batch_size=batch_size,
            cancel=cancel,
        )
        diagnostics["mapped_cap_errors"] = caps
        diagnostics["sampled_surface_error"] = max([*errors, *caps.values()])
    diagnostics["workflow_seconds"] = time.perf_counter() - start
    _check_cancel(cancel)
    return Mesh(
        **result,
        boundary_tags=tags,
        periodic_pairs=pairs,
        periodic_transforms=transforms,
        constraint_names=tuple(constraints),
    )

    seed: Mesh,
    constraints: Mapping[str, Scalar],
    *,
    bounds: Sequence[float],
    constraint_masks: npt.ArrayLike | None = None,
    seed_tolerance: float = 0.01,
    geometry_tolerance: float = 0.01,
    minimum_quality: float = 0.1,
    min_length: float | None = None,
    max_length: float | None = None,
    gradation: float = 0.3,
    adapt_iterations: int = 4,
    quality_guard: float = 0.3,
    adapt_tolerance: float | None = None,
    optimize_passes: int = 4,
    max_tetrahedra: int = 250_000,
    compile: bool = True,
    batch_size: int = 4096,
    cancel: _meshers.CancellationToken | None = None,
) -> Mesh:
    """Adapt a nonperiodic Tet4 seed against named physical constraints.

    The seed must pass `seed_tolerance`. The returned mesh must independently
    pass `geometry_tolerance` and `minimum_quality`. Constraint-mask bit j refers
    to the jth supplied field. Explicit masks allow parameter-domain cap labels
    to be replaced by their physical counterparts. No input arrays are modified.

    Returns:
        A physical Mesh with retained constraint names and new vertex masks.

    Raises:
        ValueError: Input arrays, labels, bounds or options are invalid.
        MeshingError: A seed or final geometry, topology or quality gate fails.
        CancelledError: The cancellation token is set.
    """
    from . import Mesh

    b, _, _ = _validate_options(
        constraints,
        bounds=bounds,
        cells=24,
        periodic=(False, False, False),
        coordinate_map=None,
        periodic_transforms=None,
        optimize_passes=optimize_passes,
        snap=0,
        minimum_quality=minimum_quality,
        geometry_tolerance=geometry_tolerance,
    )
    if seed.tetrahedra.ndim != 2 or seed.tetrahedra.shape[1] != 4:
        raise ValueError("seed must contain Tet4 cells")
    if any(len(pairs) for pairs in seed.periodic_pairs):
        raise ValueError("seed adaptation requires nonperiodic physical geometry")
    if constraint_masks is None:
        if seed.constraint_masks is None or seed.constraint_names != tuple(constraints):
            raise ValueError("seed constraint names must match the supplied physical fields")
        constraint_masks = seed.constraint_masks
    masks = np.asarray(constraint_masks)
    if (
        masks.shape != (len(seed.points),)
        or masks.dtype.kind not in "iu"
        or np.any(masks < 0)
        or np.any(masks >= 2**_meshers.MAX_CONSTRAINTS)
    ):
        raise ValueError("seed constraint masks must be bounded integer bits, one per vertex")
    cells = np.asarray(seed.tetrahedra)
    if cells.dtype.kind not in "iu" or np.any(cells < 0) or np.any(cells >= len(seed.points)):
        raise ValueError("seed connectivity must contain valid integer node indices")
    start = time.perf_counter()
    fields = [_prepare(f, compile=compile) for f in constraints.values()]
    preparation = time.perf_counter() - start
    _check_cancel(cancel)
    result = _meshers.adapt_intersection(
        fields,
        np.asarray(seed.points, dtype=np.float64),
        np.asarray(cells, dtype=np.int64),
        np.asarray(masks, dtype=np.uint32),
        [b[::2].tolist(), b[1::2].tolist()],
        geometry_tolerance,
        minimum_quality,
        max_tetrahedra,
        optimize_passes,
        (
            geometry_tolerance if min_length is None else float(min_length),
            float(np.min(b[1::2] - b[::2])) / 4 if max_length is None else float(max_length),
            float(gradation),
            0.0,
            int(adapt_iterations),
            float(quality_guard),
            geometry_tolerance if adapt_tolerance is None else float(adapt_tolerance),
        ),
        seed_tolerance,
        batch_size,
        cancel,
    )
    result["periodic_pairs"] = tuple(result["periodic_pairs"])
    result["diagnostics"].update(
        preparation_seconds=preparation,
        workflow_seconds=time.perf_counter() - start,
        evaluator="compiled"
        if all(isinstance(f, _meshers.CompiledField) for f in fields)
        else "mixed",
        threads=1,
    )
    return Mesh(**result, constraint_names=tuple(constraints))

    """Owned arrays and diagnostics for a tetrahedral mesh.

    Attributes:
        points: Float64 coordinates, shape ``(N, 3)``.
        tetrahedra: Int64 connectivity, shape ``(M, 4)`` or ``(M, 10)``.
        surface: Boundary triangles, shape ``(K, 3)`` or ``(K, 6)``.

        constraint_names: Names corresponding to the constraint-mask bits.
        periodic_transforms: Three low-to-high rigid 4x4 transforms for intersections.
            For mapped meshes, use these to rotate vector degrees of freedom; the
            exported PeriodicShift alone only describes point displacement.

    Arrays remain mutable even though the container is frozen. Editing coordinates
    or connectivity can invalidate quality values and periodic metadata.

    tetrahedra: npt.NDArray[np.int64]
    surface: npt.NDArray[np.int64]
    boundary_tags: npt.NDArray[np.uint8]
    periodic_pairs: tuple[npt.NDArray[np.int64], ...]

    constraint_masks: npt.NDArray[np.uint8] | npt.NDArray[np.uint32] | None = None
    constraint_names: tuple[str, ...] = ()
    periodic_transforms: npt.NDArray[np.float64] | None = None

    def surface_for(self, name: str) -> npt.NDArray[np.int64]:
        """Return boundary triangles assigned to a named intersection constraint.

        Returns:
            Triangle connectivity referencing ``points``.

        Raises:
            ValueError: The mesh has no constraint with this name.
        """
        if name not in self.constraint_names or self.constraint_masks is None:
            raise ValueError(f"Unknown constraint: {name}")
        bit = 1 << self.constraint_names.index(name)
        masks = self.constraint_masks[self.surface]
        labels = masks[:, 0] & masks[:, 1] & masks[:, 2]
        return self.surface[(labels & bit) != 0]

    @property
    def feature_edges(self) -> npt.NDArray[np.int64]:
        """Boundary edges shared by at least two named constraints."""
        if self.constraint_masks is None:
            return np.empty((0, 2), dtype=np.int64)
        edges = np.unique(
            np.sort(
                np.concatenate([
                    self.surface[:, [0, 1]],
                    self.surface[:, [1, 2]],
                    self.surface[:, [2, 0]],
                ]),
                axis=1,
            ),
            axis=0,
        )
        common = self.constraint_masks[edges[:, 0]] & self.constraint_masks[edges[:, 1]]
        return edges[(common & (common - np.uint8(1))) != 0]

        Linear cells retain the Volume and MMGQuality arrays.

def compile_field(
    function: Callable[..., object],
    *,
    gradient: Callable[..., object] | Literal[False] | None = None,
    backend: str = "rust",
) -> _meshers.CompiledField:
    """Prepare an immutable native evaluator. Usually called automatically.

    Args:
        function: Pointwise NumPy-style function of x, y, z.
        gradient: None differentiates the expression graph automatically. A callable
            returns three scalar expressions; False retains finite differences.
        backend: ``rust`` or its alias ``jit``. Both use the embedded compiler.

    Returns:
        CompiledField: A cached float64 evaluator owning its executable memory.
            Captured constants are frozen; preparing again observes changed parameters.

    Raises:
        TraceError: The callable uses unsupported operations or exceeds the graph limit.
        ValueError: The backend name or a constant is invalid.

    Note:
        No Rust installation, Numba, SymPy or subprocess is used at runtime.
        Gradients at abs(0) are zero. Exact min/max ties average the operand
        derivatives. This is a subgradient choice at a true crease, not a
        branch constraint. Eligible meshing modes can additionally retain
        outer-maximum branches. Adaptive Tet4 meshing can also retain the
        zero-sublevel intersection of a standard clamped-norm signed-distance
        graph with unit-distance affine/radial branches, after numerical checks.
        Supply a gradient callable for another policy.
    """
    if backend not in {"rust", "jit"}:
        raise ValueError("only the embedded Rust compiler is supported")
    graph = _Graph()
    xyz = [graph._intern(("input", a)) for a in ("x", "y", "z")]
    value = graph.convert(function(*xyz))
    outputs = [value]
    if gradient is None:
        outputs += graph.derivatives(value)
    elif gradient is not False:
        g = gradient(*xyz)
        if not isinstance(g, (tuple, list)) or len(g) != 3:
            raise TraceError("gradient must return three scalar expressions")
        outputs.extend(graph.convert(v) for v in g)
    if len(graph.nodes) > 20000:
        raise TraceError("expression is too large to compile")
    outputs = tuple(e.index for e in outputs)
    return (
        _compile_with_intersection(graph, value, outputs)
        if gradient is None
        else _compile_cached(graph.encoded(), outputs)
    )
