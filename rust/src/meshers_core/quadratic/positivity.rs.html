1//! Bound the determinant on a reference tet using the affine variation of its Jacobian.
2use super::jacobian_matrix;
3use crate::{Point, cross, dot};
4
5type Matrix = [Point; 3];
6
7/// The input coordinates are relative to the first node, including when fitting.
8pub(super) fn positive(x: &[Point; 10]) -> bool {
9    let corners = std::array::from_fn(|i| {
10        let mut l = [0.; 4];
11        l[i] = 1.;
12        jacobian_matrix(x, l)
13    });
14    // Scale per physical coordinate, allowing for cancellation in the quadratic
15    // derivatives and in the triple products below. Keep this guard on subdivision.
16    let scales: Point =
17        std::array::from_fn(|axis| x.iter().map(|p| p[axis].abs()).fold(0., f64::max));
18    let guard = 4096. * f64::EPSILON * scales.iter().product::<f64>();
19    let mut budget = 4096;
20    bound(&corners, guard, 0, &mut budget)
21}
22
23fn bound(corners: &[Matrix; 4], guard: f64, depth: usize, budget: &mut usize) -> bool {
24    if *budget == 0 {
25        return false;
26    }
27    *budget -= 1;
28    if corners.iter().any(|m| {
29        let d = dot(m[0], cross(m[1], m[2]));
30        !d.is_finite() || d <= guard
31    }) {
32        return false;
33    }
34    // Each row is a convex combination of its four corner rows. By
35    // multilinearity the determinant is a convex combination of these 64
36    // triple products (even when the barycentric weights are shared).
37    let mut mixed = [[[0.; 4]; 4]; 4];
38    for (b, jb) in corners.iter().enumerate() {
39        for (c, jc) in corners.iter().enumerate() {
40            let normal = cross(jb[1], jc[2]);
41            for (a, ja) in corners.iter().enumerate() {
42                mixed[a][b][c] = dot(ja[0], normal);
43            }
44        }
45    }
46    if mixed.iter().flatten().flatten().all(|&d| d > guard) {
47        return true;
48    }
49    if depth >= 20 {
50        return false;
51    }
52    // Split where the determinant bound varies. Shear can change the matrix
53    // substantially along an edge while leaving every triple product unchanged.
54    let mut edge = (0, 1);
55    let mut largest = 0.;
56    for i in 0..4 {
57        for j in 0..i {
58            let mut difference: f64 = 0.;
59            for a in 0..4 {
60                for b in 0..4 {
61                    difference = difference
62                        .max((mixed[i][a][b] - mixed[j][a][b]).abs())
63                        .max((mixed[a][i][b] - mixed[a][j][b]).abs())
64                        .max((mixed[a][b][i] - mixed[a][b][j]).abs());
65                }
66            }
67            if difference > largest {
68                largest = difference;
69                edge = (i, j);
70            }
71        }
72    }
73    let (i, j) = edge;
74    let midpoint = std::array::from_fn(|r| {
75        std::array::from_fn(|c| 0.5 * corners[i][r][c] + 0.5 * corners[j][r][c])
76    });
77    [i, j].into_iter().all(|k| {
78        let mut child = *corners;
79        child[k] = midpoint;
80        bound(&child, guard, depth + 1, budget)
81    })
82}

84#[cfg(test)]
85mod tests {
86    use super::*;
87    use crate::quadratic::{EDGES, jacobian, samples};
88
89    #[test]
90    fn fold_between_positive_samples_and_valid_twist() {
91        let vertices: [Point; 4] = [[0., 0., 0.], [1., 0., 0.], [0., 1., 0.], [0., 0., 1.]];
92        let nodes: [Point; 10] = std::array::from_fn(|i| {
93            if i < 4 {
94                return vertices[i];
95            }
96            let [a, b] = EDGES[i - 4];
97            std::array::from_fn(|k| (vertices[a][k] + vertices[b][k]) / 2.)
98        });
99        let folded = nodes.map(|[r, s, t]| [r, (r - 0.38) * s, (r - 0.42) * t]);
100        assert!(samples().iter().all(|&l| jacobian(&folded, l) > 0.));
101        assert!(!positive(&folded));
102        let twist =
103            nodes.map(|[r, s, t]| [r, (r - 0.4) * s - 0.025 * t, (r - 0.4) * t + 0.025 * s]);
104        assert!(positive(&twist));
105        assert!(positive(&nodes));
106    }
