1//! Bounded coupled corner repair. Commit only an entirely passing incident cavity.
2use super::{Output, ScalarField};
3use crate::implicit::Options;
4use crate::{Point, determinant, quality};
5use slsqp_rssl::{Constraint, Slsqp, SlsqpMode};
6use std::collections::{BTreeSet, HashMap};
7
8pub(super) fn improve(
9    output: &mut Output,
10    o: &Options,
11    fields: &[&dyn ScalarField],
12    map: &dyn Fn(Point) -> Point,
13) -> Result<usize, String> {
14    if o.minimum_quality == 0. || o.optimize_passes == 0 || o.periodic.iter().any(|&x| x) {
15        return Ok(0);
16    }
17    let mut bad: Vec<_> = (0..output.mesh.tets.len())
18        .filter(|&i| {
19            quality(output.mesh.tets[i].map(|v| output.mesh.points[v])).powf(1.5)
20                < o.minimum_quality
21        })
22        .collect();
23    bad.sort_by(|&a, &b| {
24        quality(output.mesh.tets[a].map(|v| output.mesh.points[v]))
25            .total_cmp(&quality(output.mesh.tets[b].map(|v| output.mesh.points[v])))
26    });
27    let neighborhoods: Vec<Vec<usize>> = bad
28        .into_iter()
29        .take(32)
30        .map(|ti| output.mesh.tets[ti].to_vec())
31        .collect();
32    project_neighborhoods(output, o, fields, map, &neighborhoods, None)
33}

35pub(crate) fn project_neighborhoods(
36    output: &mut Output,
37    o: &Options,
38    fields: &[&dyn ScalarField],
39    map: &dyn Fn(Point) -> Point,
40    neighborhoods: &[Vec<usize>],
41    geometry: Option<f64>,
42) -> Result<usize, String> {
43    let target = (o.minimum_quality + 2e-7).min(1.);
44    let cell: Point =
45        std::array::from_fn(|a| (o.bounds[1][a] - o.bounds[0][a]) / o.cells[a] as f64);
46    let h = cell.into_iter().fold(f64::INFINITY, f64::min);
47    let mut incident = vec![Vec::new(); output.mesh.points.len()];
48    for (i, t) in output.mesh.tets.iter().enumerate() {
49        for &v in t {
50            incident[v].push(i);
51        }
52    }
53    let mut moved = 0;
54    for vertices in neighborhoods.iter().take(32) {
55        for f in fields {
56            f.check()?;
57        }
58        let mut near: BTreeSet<_> = vertices.iter().copied().collect();
59        for _ in 0..2 {
60            near = near
61                .iter()
62                .flat_map(|&v| incident[v].iter().flat_map(|&i| output.mesh.tets[i]))
63                .collect();
64        }
65        // ponytail: bound dense local work; larger cavities need a sparse repair.
66        if near.len() > 64 {
67            continue;
68        }
69        let near: Vec<_> = near.into_iter().collect();
70        let ids: BTreeSet<_> = near
71            .iter()
72            .flat_map(|&v| incident[v].iter().copied())
73            .collect();
74        if ids.len() > 512 {
75            continue;
76        }
77        let lookup: HashMap<_, _> = near.iter().enumerate().map(|(j, &v)| (v, j)).collect();
78        let guarded: Vec<_> = if geometry.is_some() {
79            output
80                .mesh
81                .surface
82                .iter()
83                .filter(|f| f.iter().any(|v| lookup.contains_key(v)))
84                .map(|&f| {
85                    let active = f.iter().fold(super::ConstraintMask::MAX, |mask, &v| {
86                        mask & output.constraints[v]
87                    });
88                    (f, active)
89                })
90                .collect()
91        } else {
92            Vec::new()
93        };
94        let origins: Vec<_> = near.iter().map(|&v| output.parameters[v]).collect();
95        let active: Vec<Vec<usize>> = near
96            .iter()
97            .map(|&v| {
98                (0..fields.len())
99                    .filter(|&j| output.constraints[v] & (1 << j) != 0)
100                    .collect()
101            })
102            .collect();
103        let locked: Vec<[bool; 3]> = origins
104            .iter()
105            .map(|p| std::array::from_fn(|a| p[a] == o.bounds[0][a] || p[a] == o.bounds[1][a]))
106            .collect();
107        let retract = |a: &[Point]| -> Option<Vec<Point>> {
108            a.iter()
109                .enumerate()
110                .map(|(j, &p)| {
111                    let mask = active[j].iter().fold(0u64, |mask, &i| mask | (1u64 << i));
112                    let mut candidate = p;
113                    for a in 0..3 {
114                        if locked[j][a] {
115                            candidate[a] = origins[j][a];
116                        }
117                    }
118                    let p = crate::projection::onto_constraints(
119                        fields, mask, candidate, locked[j], o.bounds,
120                    )?;
121                    if (0..3).any(|a| p[a] < o.bounds[0][a] || p[a] > o.bounds[1][a])
122                        || fields.iter().any(|&f| {
123                            let value = f.value(p);
124                            value > 0.
125                                && value
126                                    > 1e-10 * crate::norm(super::field_gradient(f, p, h * 1e-5))
127                        })
128                    {
129                        None
130                    } else {
131                        Some(p)
132                    }
133                })
134                .collect()
135        };
136        let physical = |a: &[Point]| a.iter().map(|&p| map(p)).collect::<Vec<_>>();
137        let passing = |phys: &[Point]| {
138            ids.iter().all(|&i| {
139                let tet = output.mesh.tets[i]
140                    .map(|v| lookup.get(&v).map_or(output.mesh.points[v], |&j| phys[j]));
141                let det = determinant(tet);
142                det.is_finite() && det > 0. && quality(tet).powf(1.5) >= o.minimum_quality
143            }) && guarded.iter().all(|&(face, active)| {
144                let p = face.map(|v| lookup.get(&v).map_or(output.mesh.points[v], |&j| phys[j]));
145                crate::projection::facet_distance(fields, u64::from(active), p, o.bounds)
146                    <= geometry.unwrap()
147            })
148        };
149        let initial = retract(&origins).filter(|a| passing(&physical(a)));
150        let a = if let Some(a) = initial {
151            a
152        } else {
153            let mut variables = vec![[None; 3]; near.len()];
154            let mut bounds = Vec::new();
155            for j in 0..near.len() {
156                for axis in 0..3 {
157                    if !locked[j][axis] {
158                        variables[j][axis] = Some(bounds.len());
159                        bounds.push((
160                            (-1_f64).max((o.bounds[0][axis] - origins[j][axis]) / cell[axis]),
161                            1_f64.min((o.bounds[1][axis] - origins[j][axis]) / cell[axis]),
162                        ));
163                    }
164                }
165            }
166            if bounds.is_empty() {
167                continue;
168            }
169            let point = |j: usize, x: &[f64]| -> Point {
170                std::array::from_fn(|axis| {
171                    origins[j][axis] + variables[j][axis].map_or(0., |k| x[k] * cell[axis])
172                })
173            };
174            let point = &point;
175            let mut constraints: Vec<Constraint<'_>> = Vec::new();
176            for &i in &ids {
177                let tet = output.mesh.tets[i];
178                let lookup = &lookup;
179                let reference = &output.mesh.points;
180                constraints.push(Constraint::Ineq(Box::new(move |x| {
181                    let p = tet.map(|v| lookup.get(&v).map_or(reference[v], |&j| map(point(j, x))));
182                    let q = quality(p).powf(1.5) * determinant(p).signum();
183                    if q.is_finite() { q - target } else { -1. }
184                })));
185            }
186            for &(face, active) in &guarded {
187                let lookup = &lookup;
188                let reference = &output.mesh.points;
189                constraints.push(Constraint::Ineq(Box::new(move |x| {
190                    let p =
191                        face.map(|v| lookup.get(&v).map_or(reference[v], |&j| map(point(j, x))));
192                    let error =
193                        crate::projection::facet_distance(fields, u64::from(active), p, o.bounds);
194                    // Optimize inside the gate; solver residuals do not admit a mesh.
195                    if error.is_finite() {
196                        0.9999 - error / geometry.unwrap()
197                    } else {
198                        -1.
199                    }
200                })));
201            }
202            for j in 0..near.len() {
203                let mut basis: Vec<Point> = Vec::new();
204                for (i, &field) in fields.iter().enumerate() {
205                    let gradient = super::field_gradient(field, origins[j], h * 1e-5);
206                    let length = crate::norm(gradient);
207                    let scale = if length.is_finite() && length > 0. {
208                        length
209                    } else {
210                        let value = field.value(origins[j]).abs();
211                        if value > 0. && value.is_finite() {
212                            value
213                        } else {
214                            1.
215                        }
216                    };
217                    let is_active = active[j].contains(&i);
218                    let mut normal = gradient.map(|g| g / scale);
219                    for (value, locked) in normal.iter_mut().zip(locked[j]) {
220                        if locked {
221                            *value = 0.;
222                        }
223                    }
224                    for &n in &basis {
225                        normal = crate::sub(normal, n.map(|v| v * crate::dot(normal, n)));
226                    }
227                    let rank = crate::norm(normal);
228                    if is_active && rank.is_finite() && rank > 1e-10 {
229                        basis.push(normal.map(|v| v / rank));
230                        constraints.push(Constraint::Eq(Box::new(move |x| {
231                            field.value(point(j, x)) / scale
232                        })));
233                    } else {
234                        constraints.push(Constraint::Ineq(Box::new(move |x| {
235                            -field.value(point(j, x)) / scale
236                        })));
237                        if is_active {
238                            constraints.push(Constraint::Ineq(Box::new(move |x| {
239                                field.value(point(j, x)) / scale
240                            })));
241                        }
242                    }
243                }
244            }
245            let iterations = 500;
246            let mut solver = Slsqp::new_with_observer(
247                vec![0.; bounds.len()],
248                &bounds,
249                Box::new(|x| 0.5 * x.iter().map(|v| v * v).sum::<f64>()),
250                constraints,
251                iterations,
252                1e-12,
253                (),
254            );
255            for _ in 0..(3 * iterations + 1) {
256                for field in fields {
257                    field.check()?;
258                }
259                match solver.step() {
260                    Ok(SlsqpMode::Success) | Err(_) => break,
261                    Ok(_) => {}
262                }
263            }
264            for field in fields {
265                field.check()?;
266            }
267            let parameters: Vec<_> = (0..near.len()).map(|j| point(j, &solver.x)).collect();
268            let Some(a) = retract(&parameters) else {
269                continue;
270            };
271            a
272        };
273        let phys = physical(&a);
274        if passing(&phys) {
275            for (j, &v) in near.iter().enumerate() {
276                moved +=
277                    usize::from(output.parameters[v] != a[j] || output.mesh.points[v] != phys[j]);
278                output.parameters[v] = a[j];
279                output.mesh.points[v] = phys[j];
280            }
281        }
282    }
283    Ok(moved)
284}

304    fn check_corner(input: &str, cells: [usize; 3]) {
305        for scale in [1e-12, 1., 1e12] {
306            check_scaled_corner(input, cells, scale);
307        }

309    fn check_scaled_corner(input: &str, cells: [usize; 3], scale: f64) {
310        #[derive(serde::Deserialize)]
311        struct Star {
312            points: Vec<Point>,
313            parameters: Vec<Point>,
314            tets: Vec<crate::Tet>,
315            constraints: Vec<super::super::ConstraintMask>,
316        }
317        let seed: Star = serde_json::from_str(input).unwrap();
318        let gyroid = |p: Point| {
319            let a = p.map(|v| v * std::f64::consts::TAU);
320            a[0].sin() * a[1].cos() + a[1].sin() * a[2].cos() + a[2].sin() * a[0].cos()
321        };
322        let upper = |p: Point| scale * (gyroid(p) - 0.55);
323        let lower = |p: Point| scale * (-gyroid(p) - 0.55);
324        let map = |p: Point| {
325            let angle = std::f64::consts::TAU * (p[2] + 0.5);
326            let r = 0.3 + 0.2 * p[1];
327            [r * angle.cos(), r * angle.sin(), p[0] + 0.5]
328        };
329        let mut out = Output {
330            mesh: crate::Mesh {
331                points: seed.points,
332                tets: seed.tets,
333                surface: Vec::new(),
334            },
335            parameters: seed.parameters,
336            constraints: seed.constraints,
337            quality: Default::default(),
338        };
339        let original_points = out.mesh.points.clone();
340        let original_parameters = out.parameters.clone();
341        let original_masks = out.constraints.clone();
342        let original_tets = out.mesh.tets.clone();
343        let old = out
344            .mesh
345            .tets
346            .iter()
347            .map(|t| quality(t.map(|v| out.mesh.points[v])).powf(1.5))
348            .fold(1., f64::min);
349        assert!(old < 0.1);
350        let o = Options {
351            bounds: [[-0.5, -0.5, -1.], [0.5, 0.5, 1.]],
352            cells,
353            minimum_quality: 0.1,
354            ..Default::default()
355        };
356        assert_eq!(
357            improve(
358                &mut out,
359                &Options {
360                    optimize_passes: 0,
361                    ..o
362                },
363                &[&upper, &lower],
364                &map
365            )
366            .unwrap(),
367            0
368        );
369        assert_eq!(out.mesh.points, original_points);
370        assert_eq!(out.parameters, original_parameters);
371        assert!(
372            improve(&mut out, &o, &[&upper, &lower], &map).unwrap() > 0,
373            "corner grid {cells:?}, initial quality {old}"
374        );
375        assert!(
376            out.mesh
377                .tets
378                .iter()
379                .all(|t| determinant(t.map(|v| out.mesh.points[v])) > 0.
380                    && quality(t.map(|v| out.mesh.points[v])).powf(1.5) >= 0.1)
381        );
382        assert_eq!(out.constraints, original_masks);
383        assert_eq!(out.mesh.tets, original_tets);
384        for (i, &parameter) in out.parameters.iter().enumerate() {
385            for axis in 0..3 {
386                if original_parameters[i][axis] == o.bounds[0][axis]
387                    || original_parameters[i][axis] == o.bounds[1][axis]
388                {
389                    assert_eq!(parameter[axis], original_parameters[i][axis]);
390                }
391            }
392            for (bit, field) in [&upper as &dyn ScalarField, &lower as &dyn ScalarField]
393                .into_iter()
394                .enumerate()
395            {
396                if out.constraints[i] & (1 << bit) != 0 {
397                    let before = field.value(original_parameters[i]).abs() / scale;
398                    let after = field.value(parameter).abs() / scale;
399                    assert!(after <= before.max(1e-9));
400                    if parameter != original_parameters[i]
401                        || out.mesh.points[i] != original_points[i]
402                    {
403                        assert!(after < 1e-9);
404                    }
405                }
406            }
407            if parameter != original_parameters[i] || out.mesh.points[i] != original_points[i] {
408                assert!(crate::norm(crate::sub(map(parameter), out.mesh.points[i])) < 1e-12);
409            }
410        }
411    }
