1//! Internal geometry context shared by specialized and generic algorithms.
2use crate::Point;
3pub(crate) trait Geometry: Sync {
4    fn constraints(&self) -> Option<&[&dyn crate::implicit::ScalarField]> {
5        None
6    }
7    fn halfspaces(&self) -> Option<&[[f64; 4]]> {
8        None
9    }
10    fn halfspace_regions(&self) -> Option<&[crate::implicit::intersection::ConstraintMask]> {
11        None
12    }
13    /// Physical reference length for dimensionless optimizer penalties.
14    fn optimizer_length(&self) -> f64 {
15        1.
16    }
17    fn refinement_quality_floor(&self) -> f64 {
18        0.25
19    }
20    fn minimum_quality(&self) -> f64 {
21        0.
22    }
23    fn value(&self, p: Point) -> f64;
24    fn batched(&self) -> bool {
25        false
26    }
27    fn values(&self, points: &[Point]) -> Vec<f64> {
28        points.iter().map(|&p| self.value(p)).collect()
29    }
30    fn target_lengths(&self, _points: &[Point]) -> Option<Vec<f64>> {
31        None
32    }
33    fn gradients(&self, points: &[Point]) -> Vec<Point> {
34        points.iter().map(|&p| self.gradient(p)).collect()
35    }
36    fn check(&self) -> Result<(), String> {
37        Ok(())
38    }
39    fn gradient(&self, p: Point) -> Point;
40    fn rank(&self, p: Point) -> [i64; 3];
41    fn bounds(&self) -> [Point; 2];
42    fn band(&self) -> bool {
43        true
44    }
45    /// Implicit pipelines preserve this limit while improving element quality.
46    fn geometry_tolerance(&self) -> Option<f64> {
47        None
48    }
49    fn locked(&self, p: Point) -> [bool; 3] {
50        let [lo, hi] = self.bounds();
51        std::array::from_fn(|a| p[a] == lo[a] || p[a] == hi[a])
52    }
53    fn contains(&self, p: Point) -> bool {
54        let [lo, hi] = self.bounds();
55        (0..3).all(|a| p[a] >= lo[a] && p[a] <= hi[a])
56    }
57    fn residual(&self, p: Point, t: f64) -> f64 {
58        let v = self.value(p);
59        if self.band() { v.abs() - t } else { v }
60    }
61    fn level(&self, value: f64, t: f64) -> f64 {
62        if self.band() { value.signum() * t } else { 0. }
63    }
64    fn cap(&self, points: [Point; 3]) -> bool {
65        let [lo, hi] = self.bounds();
66        (0..3).any(|a| points.iter().all(|p| p[a] == lo[a]) || points.iter().all(|p| p[a] == hi[a]))
67    }
68}
